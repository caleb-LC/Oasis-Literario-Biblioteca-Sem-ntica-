Consulta de ejemplo

SELECT DISTINCT ?tituloLibro ?tipoUsuario WHERE { ?usuario ont:nombre ?nombreUsuario . FILTER(regex(str(?nombreUsuario), "Juan", "i")) ?usuario rdf:type ?tipoURI . BIND(STRAFTER(STR(?tipoURI), "#") AS ?tipoUsuario) FILTER(?tipoUsuario IN ("Estudiante","Investigador","Profesor")) ?usuario ont:realizaPrestamo ?prestamo . ?prestamo ont:incluyeRecurso ?libro . ?libro ont:tieneTitulo ?tituloLibro . }
